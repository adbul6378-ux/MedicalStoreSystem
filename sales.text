    <script>
        // Sample Medicine Data
        let cartItems = [
            { id: 1, name: 'Paracetamol 500mg (Tab)', batch: 'PCT500-0525', expiry: '05/2026', mrp: 15.00, price: 10.00, qty: 2, discount: 0 },
            { id: 2, name: 'Amoxicillin 500mg (Cap)', batch: 'AMX500-0326', expiry: '03/2026', mrp: 30.00, price: 22.00, qty: 1, discount: 0 },
            { id: 3, name: 'Omeprazole 20mg (Cap)', batch: 'OMP20-0626', expiry: '06/2026', mrp: 25.00, price: 18.00, qty: 2, discount: 0 },
            { id: 4, name: 'Vitamin C 500mg (Tab)', batch: 'VC500-0726', expiry: '07/2026', mrp: 20.00, price: 15.00, qty: 1, discount: 0 },
            { id: 5, name: 'Cetirizine 10mg (Tab)', batch: 'CET10-0526', expiry: '05/2026', mrp: 18.00, price: 12.00, qty: 1, discount: 0 }
        ];

        // Render Cart Items & Update Totals
        function renderCart() {
            const tbody = document.getElementById('cartTableBody');
            const receiptBody = document.getElementById('receiptItems');
            
            tbody.innerHTML = '';
            receiptBody.innerHTML = '';

            let subtotal = 0;

            cartItems.forEach((item, index) => {
                const itemTotal = (item.price * item.qty) - item.discount;
                subtotal += itemTotal;

                // Main Table Row
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td>${index + 1}</td>
                    <td style="font-weight: 500;">${item.name}</td>
                    <td style="color: var(--text-muted);">${item.batch}</td>
                    <td style="color: var(--text-muted);">${item.expiry}</td>
                    <td>${item.mrp.toFixed(2)}</td>
                    <td>${item.price.toFixed(2)}</td>
                    <td><input type="number" class="input-qty" value="${item.qty}" min="1" onchange="updateQty(${item.id}, this.value)"></td>
                    <td><input type="number" class="input-discount" value="${item.discount.toFixed(2)}" min="0" onchange="updateDiscount(${item.id}, this.value)"></td>
                    <td style="font-weight: 700;">${itemTotal.toFixed(2)}</td>
                    <td><button class="btn-delete" onclick="removeItem(${item.id})"><i class="fa-regular fa-trash-can"></i></button></td>
                `;
                tbody.appendChild(tr);

                // Thermal Receipt Row
                const rTr = document.createElement('tr');
                rTr.innerHTML = `
                    <td>${index + 1}</td>
                    <td>${item.name}</td>
                    <td style="text-align: center;">${item.qty}</td>
                    <td style="text-align: right;">${item.price.toFixed(2)}</td>
                    <td style="text-align: right;">${itemTotal.toFixed(2)}</td>
                `;
                receiptBody.appendChild(rTr);
            });

            // Calculate overall totals
            const discountPercent = parseFloat(document.getElementById('discountInput').value) || 0;
            const taxPercent = parseFloat(document.getElementById('taxInput').value) || 0;

            const discountAmount = (subtotal * discountPercent) / 100;
            const taxAmount = ((subtotal - discountAmount) * taxPercent) / 100;
            const grandTotal = subtotal - discountAmount + taxAmount;

            // Update UI Total elements
            document.getElementById('subtotalVal').innerText = `Rs. ${subtotal.toFixed(2)}`;
            document.getElementById('discountVal').innerText = `Rs. ${discountAmount.toFixed(2)}`;
            document.getElementById('taxVal').innerText = `Rs. ${taxAmount.toFixed(2)}`;
            document.getElementById('grandTotalVal').innerText = `Rs. ${grandTotal.toFixed(2)}`;

            // Thermal Receipt Totals
            document.getElementById('rSubtotal').innerText = `Rs. ${subtotal.toFixed(2)}`;
            document.getElementById('rDiscount').innerText = `Rs. ${discountAmount.toFixed(2)}`;
            document.getElementById('rTax').innerText = `Rs. ${taxAmount.toFixed(2)}`;
            document.getElementById('rTotal').innerText = `Rs. ${grandTotal.toFixed(2)}`;

            // Change Return Calculation
            const paidVal = parseFloat(document.getElementById('paidInput').value) || 0;
            const change = paidVal - grandTotal;
            document.getElementById('changeVal').value = change >= 0 ? change.toFixed(2) : '0.00';
            
            document.getElementById('rPaid').innerText = `Rs. ${paidVal.toFixed(2)}`;
            document.getElementById('rChange').innerText = `Rs. ${(change >= 0 ? change : 0).toFixed(2)}`;
        }

        // Functions for Interactive Table
        function updateQty(id, newQty) {
            const item = cartItems.find(i => i.id === id);
            if (item) {
                item.qty = parseInt(newQty) || 1;
                renderCart();
            }
        }

        function updateDiscount(id, newDisc) {
            const item = cartItems.find(i => i.id === id);
            if (item) {
                item.discount = parseFloat(newDisc) || 0;
                renderCart();
            }
        }

        function removeItem(id) {
            cartItems = cartItems.filter(i => i.id !== id);
            renderCart();
        }

        document.getElementById('btnClearAll').addEventListener('click', () => {
            cartItems = [];
            renderCart();
        });

        // Event listeners for Inputs
        document.getElementById('discountInput').addEventListener('input', renderCart);
        document.getElementById('taxInput').addEventListener('input', renderCart);
        document.getElementById('paidInput').addEventListener('input', renderCart);

        // Initial Load
        renderCart();
    </script>
