<script>
let editRow=null;
function toggleSidebar(){document.querySelector('.sidebar').classList.toggle('collapsed')}
function openModal(row=null){
 editRow=row;
 document.getElementById('modalTitle').textContent=row?'Edit Teacher':'Add Teacher';
 if(row){
   let c=row.cells;
   document.getElementById('fName').value=c[2].textContent.trim();
   document.getElementById('fDesignation').value=c[3].textContent.trim();
   document.getElementById('fSubject').value=c[4].textContent.trim();
   document.getElementById('fDept').value=c[5].textContent.trim();
   document.getElementById('fPhone').value=c[6].textContent.trim();
 }else document.querySelectorAll('.modal input').forEach(x=>x.value='');
 document.getElementById('teacherModal').classList.add('show');
}
function closeModal(){document.getElementById('teacherModal').classList.remove('show')}
function saveTeacher(){
 const name=document.getElementById('fName').value.trim();
 if(!name){alert('Please enter teacher name.');return}
 const des=document.getElementById('fDesignation').value||'Teacher', sub=document.getElementById('fSubject').value||'—';
 const dept=document.getElementById('fDept').value, phone=document.getElementById('fPhone').value||'Not provided', cls=document.getElementById('fClass').value;
 if(editRow){
   editRow.cells[2].textContent=name;editRow.cells[3].textContent=des;editRow.cells[4].textContent=sub;editRow.cells[5].textContent=dept;editRow.cells[6].textContent=phone;
   editRow.dataset.name=name;editRow.dataset.dept=dept;editRow.dataset.class=cls;
 }else{
   const tbody=document.querySelector('#teacherTable tbody'), row=tbody.insertRow();
   row.dataset.name=name;row.dataset.dept=dept;row.dataset.class=cls;
   const n=tbody.rows.length;
   row.innerHTML=`<td>${n}</td><td><div class="photo">👨‍🏫</div></td><td class="name">${name}</td><td>${des}</td><td>${sub}</td><td>${dept}</td><td>${phone}</td><td><span class="status">Active</span></td><td><div class="actions"><button class="act" onclick="viewTeacher(this)">◉</button><button class="act edit" onclick="editTeacher(this)">✎</button><button class="act del" onclick="deleteTeacher(this)">♟</button></div></td>`;
 }
 closeModal();alert(editRow?'Teacher updated successfully.':'Teacher added successfully.');
 editRow=null;
}
function editTeacher(btn){openModal(btn.closest('tr'))}
function viewTeacher(btn){
 const r=btn.closest('tr');alert(`Teacher: ${r.cells[2].textContent}\\nDesignation: ${r.cells[3].textContent}\\nSubject: ${r.cells[4].textContent}\\nDepartment: ${r.cells[5].textContent}\\nPhone: ${r.cells[6].textContent}`);
}
function deleteTeacher(btn){
 const r=btn.closest('tr');
 if(confirm(`Delete ${r.cells[2].textContent}?`)){r.remove();renumber();alert('Teacher removed.')}
}
function renumber(){document.querySelectorAll('#teacherTable tbody tr').forEach((r,i)=>r.cells[0].textContent=i+1)}
function filterTeachers(){
 const q=(document.getElementById('globalSearch').value+' '+document.getElementById('teacherSearch').value).toLowerCase();
 const dept=document.getElementById('deptFilter').value, cls=document.getElementById('classFilter').value;
 document.querySelectorAll('#teacherTable tbody tr').forEach(r=>{
   const matchQ=!q || r.innerText.toLowerCase().includes(q);
   const matchD=!dept || r.dataset.dept===dept;
   const matchC=!cls || r.dataset.class===cls;
   r.style.display=matchQ&&matchD&&matchC?'':'none';
 });
}
function clearFilters(){document.getElementById('globalSearch').value='';document.getElementById('teacherSearch').value='';document.getElementById('deptFilter').value='';document.getElementById('classFilter').value='';filterTeachers()}
