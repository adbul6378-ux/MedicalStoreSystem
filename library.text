<script>

/* ================= TOAST ================= */

function showToast(message){

    const toast =
        document.getElementById("toast");

    toast.innerText = message;

    toast.style.display = "block";

    clearTimeout(window.toastTimer);

    window.toastTimer =
        setTimeout(function(){

            toast.style.display = "none";

        },2500);
}


/* ================= MOBILE SIDEBAR ================= */

document
.getElementById("hamburger")
.addEventListener("click",function(){

    document
    .getElementById("sidebar")
    .classList.toggle("show");

});


/* ================= SEARCH ================= */

document
.getElementById("search")
.addEventListener("input",function(){

    const searchValue =
        this.value.toLowerCase().trim();

    filterBooks();

    const rows =
        document.querySelectorAll("#bookTable tr");

    rows.forEach(function(row){

        const text =
            row.innerText.toLowerCase();

        if(searchValue === ""){
            row.style.display = "";
        }

    });

});


/* ================= FILTER FUNCTION ================= */

function filterBooks(){

    const searchValue =
        document
        .getElementById("search")
        .value
        .toLowerCase()
        .trim();

    const category =
        document
        .getElementById("categorySelect")
        .value;

    const status =
        document
        .getElementById("statusSelect")
        .value;

    const rows =
        document.querySelectorAll("#bookTable tr");

    let visible = 0;

    rows.forEach(function(row){

        const text =
            row.innerText.toLowerCase();

        const rowCategory =
            row.dataset.category;

        const rowStatus =
            row.dataset.status;

        const searchMatch =
            text.includes(searchValue);

        const categoryMatch =
            category === "all" ||
            rowCategory === category;

        const statusMatch =
            status === "all" ||
            rowStatus === status;

        if(
            searchMatch &&
            categoryMatch &&
            statusMatch
        ){

            row.style.display = "";
            visible++;

        }

        else{

            row.style.display = "none";

        }

    });

    document
    .getElementById("bookCount")
    .innerText =
        "Showing " + visible + " books";

}


/* ================= FILTER EVENTS ================= */

document
.getElementById("categorySelect")
.addEventListener("change",filterBooks);

document
.getElementById("statusSelect")
.addEventListener("change",filterBooks);


/* ================= ADD BOOK ================= */

document
.getElementById("addBook")
.addEventListener("click",function(){

    showToast(
        "Add New Book form opened."
    );

});


document
.getElementById("quickAdd")
.addEventListener("click",function(){

    showToast(
        "Add New Book form opened."
    );

});


/* ================= ISSUE BOOK ================= */

document
.getElementById("issueBook")
.addEventListener("click",function(){

    showToast(
        "Issue Book window opened."
    );

});


/* ================= RETURN BOOK ================= */

document
.getElementById("returnBook")
.addEventListener("click",function(){

    showToast(
        "Return Book window opened."
    );

});


/* ================= OVERDUE ================= */

document
.getElementById("overdueBooks")
.addEventListener("click",function(){

    document
    .getElementById("statusSelect")
    .value = "overdue";

    filterBooks();

    showToast(
        "Showing overdue books."
    );

});


/* ================= VIEW BUTTON ================= */

document
.querySelectorAll(".viewBtn")
.forEach(function(button){

    button.addEventListener("click",function(){

        const row =
            this.closest("tr");

        const bookName =
            row.querySelector(".book-name strong")
            .innerText;

        showToast(
            "Opening details: " +
            bookName
        );

    });

});


/* ================= ISSUE / RETURN BUTTON ================= */

document
.querySelectorAll(".issueBtn")
.forEach(function(button){

    button.addEventListener("click",function(){

        const action =
            this.innerText;

        const row =
            this.closest("tr");

        const bookName =
            row.querySelector(".book-name strong")
            .innerText;

        if(action === "Issue"){

            showToast(
                bookName +
                " selected for issue."
            );

        }

        else{

            showToast(
                bookName +
                " selected for return."
            );

        }

    });

});


/* ================= SIDEBAR ================= */

document
.querySelectorAll(".menu a")
.forEach(function(link){

    link.addEventListener("click",function(e){

        e.preventDefault();

        document
        .querySelectorAll(".menu a")
        .forEach(function(item){

            item.classList.remove("active");

        });

        this.classList.add("active");

        const section =
            this.innerText.trim();

        if(section !== "Library"){

            showToast(
                section +
                " section selected."
            );

        }

    });

});
