  <script>
    // Initialize Lucide Icons
    lucide.createIcons();

    const settingsForm = document.getElementById('settingsForm');
    const settingsStorageKey = 'medicalStoreSettings';
    const savedSettings = JSON.parse(localStorage.getItem(settingsStorageKey) || '{}');
    Array.from(settingsForm.elements).forEach((field, index) => {
      if (savedSettings[index] !== undefined) field.value = savedSettings[index];
    });
    settingsForm.addEventListener('submit', (event) => {
      event.preventDefault();
      const values = {};
      Array.from(settingsForm.elements).forEach((field, index) => { if (field.name || field.value) values[index] = field.value; });
      localStorage.setItem(settingsStorageKey, JSON.stringify(values));
      const saveButton = document.getElementById('saveSettings');
      saveButton.textContent = 'Saved';
      setTimeout(() => { saveButton.textContent = 'Save Changes'; }, 1400);
    });
    document.getElementById('settingsPrint').addEventListener('click', () => window.print());
    document.getElementById('settingsAlerts').addEventListener('click', () => { window.location.href = 'expiry-alerts.html'; });
    document.getElementById('settingsCurrent').addEventListener('click', () => document.getElementById('settingsForm').scrollIntoView({ behavior: 'smooth' }));
    document.getElementById('settingsAddMedicine').addEventListener('click', () => { window.location.href = 'add-medicine.html'; });
    document.getElementById('settingsNewSale').addEventListener('click', () => { window.location.href = 'new-sale.html'; });
    document.getElementById('backupNow').addEventListener('click', () => {
      const backup = { settings: JSON.parse(localStorage.getItem(settingsStorageKey) || '{}'), customers: JSON.parse(localStorage.getItem('medicalStoreCustomers') || '[]'), createdAt: new Date().toISOString() };
      const link = document.createElement('a');
      link.href = URL.createObjectURL(new Blob([JSON.stringify(backup, null, 2)], { type: 'application/json' }));
      link.download = 'medical-store-backup.json';
      link.click();
      URL.revokeObjectURL(link.href);
    });
