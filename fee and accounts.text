/* =========================================
   FORMAT MONEY
========================================= */

function money(number){

    return "Rs. " +
        number.toLocaleString("en-PK");

}


/* =========================================
   RENDER TABLE
========================================= */

const feeTable =
    document.getElementById("feeTable");

function renderTable(){

    const search =
        document.getElementById("studentSearch")
        .value.toLowerCase();

    const selectedClass =
        document.getElementById("classFilter").value;


    const filtered =
        feeRecords.filter(record => {

            const classMatch =
                selectedClass === "All" ||
                record.class === selectedClass;

            const searchMatch =
                record.name.toLowerCase()
                .includes(search);

            return classMatch && searchMatch;

        });


    feeTable.innerHTML = "";


    filtered.slice(0,8).forEach((record,index)=>{

        const balance =
            record.total - record.paid;


        let statusClass = "";

        if(record.status === "Paid"){
            statusClass = "paid";
        }

        else if(record.status === "Pending"){
            statusClass = "pending-status";
        }

        else{
            statusClass = "overdue";
        }


        const row =
        document.createElement("tr");


        row.innerHTML = `

            <td>
                <input type="checkbox" class="feeCheck">
            </td>

            <td>${index + 1}</td>

            <td>
                <div class="student">

                    <div class="student-photo">
                        👤
                    </div>

                    ${record.name}

                </div>
            </td>

            <td>${record.class}</td>

            <td>${record.month}</td>

            <td class="amount">
                ${money(record.total)}
            </td>

            <td style="color:#078e68;font-weight:600">
                ${money(record.paid)}
            </td>

            <td style="color:#d98508;font-weight:600">
                ${money(balance)}
            </td>

            <td>
                <span class="${statusClass}">
                    ${record.status}
                </span>
            </td>

            <td>

                <button
                    class="action-btn"
                    onclick="viewFee(${feeRecords.indexOf(record)})">
                    👁
                </button>

            </td>

        `;

        feeTable.appendChild(row);

    });


    document.getElementById("showing").innerText =
        `Showing 1 to ${Math.min(filtered.length,8)}
         of ${filtered.length} records`;

}


/* =========================================
   VIEW FEE

    const student = feeRecords[index];

    const balance =
        student.total - student.paid;

    showToast(
        `${student.name}: Paid ${money(student.paid)},
        Balance ${money(balance)}`
    );

}
