    function applyFilters() {
      const statusValue = document.getElementById('statusFilter').value;
      const supplierValue = document.getElementById('supplierFilter').value;
      const searchValue = document.getElementById('globalSearch').value.toLowerCase();

      const filtered = purchasesData.filter(item => {
        const matchesStatus = statusValue === 'All' || item.status === statusValue;
        const matchesSupplier = supplierValue === 'All' || item.supplier === supplierValue;
        const matchesSearch = item.inv.toLowerCase().includes(searchValue) || 
                              item.supplier.toLowerCase().includes(searchValue);

        return matchesStatus && matchesSupplier && matchesSearch;
      });

      renderPurchasesTable(filtered);
