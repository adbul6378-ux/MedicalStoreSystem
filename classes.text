<script>

/* OPEN MODAL */
function openModal(){
    document.getElementById("classModal").style.display="flex";
}

/* CLOSE MODAL */
function closeModal(){
    document.getElementById("classModal").style.display="none";
}


/* ADD CLASS */
function addClass(){

    let className=document.getElementById("newClass").value;
    let section=document.getElementById("newSection").value;
    let students=document.getElementById("newStudents").value;
    let teacher=document.getElementById("newTeacher").value;
    let room=document.getElementById("newRoom").value;

    if(className==="" || students==="" || teacher==="" || room===""){
        alert("Please fill all fields.");
        return;
    }

    let table=document.getElementById("classTable");

    let row=table.insertRow();

    row.innerHTML=`
        <td>${table.rows.length}</td>
        <td class="class-name">${className}</td>
        <td>${section}</td>
        <td>${students}</td>
        <td>${teacher}</td>
        <td>${room}</td>
        <td><span class="badge-active">Active</span></td>
        <td>
            <button class="action-btn view-btn"
            onclick="viewClass('${className} ${section}')">👁</button>

            <button class="action-btn edit-btn"
            onclick="editClass('${className} ${section}')">✎</button>

            <button class="action-btn delete-btn"
            onclick="deleteClass(this)">🗑</button>
        </td>
    `;

    let total=document.getElementById("totalClasses");
    total.innerText=parseInt(total.innerText)+1;

    alert("Class added successfully!");

    document.getElementById("newClass").value="";
    document.getElementById("newStudents").value="";
    document.getElementById("newTeacher").value="";
    document.getElementById("newRoom").value="";

    closeModal();
}


/* SEARCH */
function searchClasses(){

    let search=document
        .getElementById("classSearch")
        .value
        .toLowerCase();

    let rows=document
        .getElementById("classTable")
        .getElementsByTagName("tr");

    for(let row of rows){

        let text=row.innerText.toLowerCase();

        if(text.includes(search)){
            row.style.display="";
        }else{
            row.style.display="none";
        }
    }
}


/* FILTER */
function filterClasses(){

    let grade=document.getElementById("gradeFilter").value;
    let section=document.getElementById("sectionFilter").value;

    let rows=document
        .getElementById("classTable")
        .getElementsByTagName("tr");

    for(let row of rows){

        let classText=row.cells[1]?.innerText || "";
        let sectionText=row.cells[2]?.innerText || "";

        let gradeMatch=
            grade==="" ||
            classText.includes("Class "+grade);

        let sectionMatch=
            section==="" ||
            sectionText===section;

        row.style.display=
            gradeMatch && sectionMatch ? "" : "none";
    }
}


/* VIEW CLASS */
function viewClass(className){
    alert("Opening details for: " + className);
}


/* EDIT CLASS */
function editClass(className){

    let newName=prompt(
        "Enter new class name:",
        className
    );

    if(newName){
        alert(
            "Class information updated successfully!"
        );
    }
}


/* DELETE CLASS */
function deleteClass(button){

    let confirmation=confirm(
        "Are you sure you want to delete this class?"
    );

    if(confirmation){

        button.closest("tr").remove();

        let total=document.getElementById("totalClasses");

        total.innerText=
            Math.max(0,parseInt(total.innerText)-1);

        alert("Class deleted successfully!");
    }
}


/* QUICK ACTION MESSAGE */
function showMessage(message){
    alert(message);
}


/* CLOSE MODAL WHEN CLICKING OUTSIDE */
window.onclick=function(event){

    let modal=document.getElementById("classModal");

    if(event.target===modal){
        closeModal();
    }
};

</script>
