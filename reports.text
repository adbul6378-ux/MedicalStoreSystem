<script>

/* =========================
   VARIABLES
========================= */

let currentReport = "";


/* =========================
   TOAST
========================= */

function showToast(message){

    const toast =
        document.getElementById("toast");

    toast.innerText = message;

    toast.style.display = "block";

    setTimeout(function(){

        toast.style.display = "none";

    },2500);

}


/* =========================
   GENERATE REPORT
========================= */

function generateReport(){

    const type =
        document.getElementById("reportType").value;

    const className =
        document.getElementById("reportClass").value;

    const date =
        document.getElementById("reportDate").value;


    if(type === ""){

        showToast(
            "Please select a report type"
        );

        return;
    }


    currentReport =
        type +
        " - " +
        className;


    document.getElementById("modalMessage")
        .innerText =
        "Your " +
        type +
        " for " +
        className +
        " has been generated successfully.";


    document.getElementById("reportModal")
        .style.display = "flex";


    if(date !== ""){

        console.log(
            "Report Date:",
            date
        );

    }

}


/* =========================
   QUICK REPORT
========================= */

function quickReport(type){

    document.getElementById("reportType")
        .value = type;

    if(
        !document.getElementById("reportType").value
    ){

        const options =
            document.getElementById("reportType")
            .options;

        for(let i=0;i<options.length;i++){

            if(
                options[i].text
                .includes(type.replace(" Report",""))
            ){

                document.getElementById("reportType")
                    .selectedIndex = i;

                break;
            }

        }

    }

    showToast(
        type + " selected"
    );

}


/* =========================
   VIEW REPORT
========================= */

function viewReport(name){

    currentReport = name;

    document.getElementById("modalMessage")
        .innerText =
        "Opening " +
        name +
        ". The report preview is ready to view.";


    document.getElementById("reportModal")
        .style.display = "flex";

}


/* =========================
   DOWNLOAD REPORT
========================= */

function downloadReport(name){

    showToast(
        "Preparing " +
        name +
        " for download..."
    );


    setTimeout(function(){

        showToast(
            name +
            " downloaded successfully"
        );

    },1200);

}


/* =========================
   DOWNLOAD CURRENT REPORT
========================= */

function downloadCurrentReport(){

    if(currentReport === ""){

        showToast(
            "No report selected"
        );

        return;
    }


    closeModal();


    downloadReport(
        currentReport
    );

}


/* =========================
   CLOSE MODAL
========================= */

function closeModal(){

    document.getElementById("reportModal")
        .style.display = "none";

}


/* =========================
   SEARCH REPORTS
========================= */

document
    .getElementById("globalSearch")
    .addEventListener(
        "input",
        function(){

            const search =
                this.value.toLowerCase();


            const reports =
                document.querySelectorAll(
                    ".report-item"
                );


            reports.forEach(function(report){

                const text =
                    report.innerText
                    .toLowerCase();


                if(
                    text.includes(search)
                ){

                    report.style.display =
                        "flex";

                }else{

                    report.style.display =
                        "none";

                }

            });

        }
    );


/* =========================
   SIDEBAR
========================= */

document
    .querySelectorAll(".menu a")
    .forEach(function(item){

        item.addEventListener(
            "click",
            function(){

                document
                    .querySelectorAll(".menu a")
                    .forEach(function(link){

                        link.classList.remove(
                            "active"
                        );

                    });


                this.classList.add(
                    "active"
                );

            }
        );

    });


/* =========================
   CLOSE MODAL OUTSIDE
========================= */

window.addEventListener(
    "click",
    function(event){

        const modal =
            document.getElementById(
                "reportModal"
            );


        if(
            event.target === modal
        ){

            closeModal();

        }

    }
);

</script>
